package com.twinping.noc.tablet;

import android.os.Bundle;
import android.webkit.JavascriptInterface;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.view.WindowManager;
import androidx.appcompat.app.AppCompatActivity;
import org.json.JSONArray;
import org.json.JSONObject;
import java.io.BufferedReader;
import java.io.InputStreamReader;

/**
 * TwinPing NOC — Standalone Android Tablet APK Native Host
 * Thực thi trực tiếp lệnh /system/bin/ping của nhân Linux Android
 * Cấu hình mặc định:
 * - Thiết bị 1: Gateway Core Router 01 (8.8.8.8) - 4 gói / 200ms
 * - Thiết bị 2: Máy Chủ Sản Xuất 02 (LAN) (192.168.1.100) - 4 gói / 200ms
 */
public class MainActivity extends AppCompatActivity {
    private WebView webView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        // Giữ màn hình máy tính bảng luôn sáng khi giám sát NOC
        getWindow().addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON);

        webView = new WebView(this);
        setContentView(webView);

        WebSettings ws = webView.getSettings();
        ws.setJavaScriptEnabled(true);
        ws.setDomStorageEnabled(true);
        ws.setDatabaseEnabled(true);
        ws.setMediaPlaybackRequiresUserGesture(false);
        ws.setAllowFileAccess(true);

        webView.addJavascriptInterface(new NativePingBridge(), "AndroidNativePing");
        webView.setWebViewClient(new WebViewClient());

        // Tải giao diện đóng gói cục bộ trong APK (assets/public/index.html)
        webView.loadUrl("file:///android_asset/public/index.html");
    }

    public class NativePingBridge {
        @JavascriptInterface
        public String executeNativePing(String ipAddress, int packetCount, int intervalMs, int timeoutMs) {
            try {
                float intervalSec = Math.max(0.2f, intervalMs / 1000.0f);
                int timeoutSec = Math.max(1, timeoutMs / 1000);
                String cmd = String.format(
                    "/system/bin/ping -c %d -i %.2f -W %d %s",
                    packetCount, intervalSec, timeoutSec, ipAddress
                );

                Process process = Runtime.getRuntime().exec(cmd);
                BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()));
                StringBuilder output = new StringBuilder();
                String line;
                while ((line = reader.readLine()) != null) {
                    output.append(line).append("\n");
                }
                int exitCode = process.waitFor();

                JSONObject result = new JSONObject();
                result.put("exitCode", exitCode);
                result.put("rawOutput", output.toString());
                result.put("ipAddress", ipAddress);
                return result.toString();
            } catch (Exception e) {
                return "{\"error\":\"" + e.getMessage() + "\"}";
            }
        }
    }
}

[3] FILE: .github/workflows/build-apk.yml (TỰ ĐỘNG XUẤT FILE .APK TRÊN GITHUB)
--------------------------------------------------------------------------------
name: Build TwinPing Android APK

on:
  workflow_dispatch:
  push:
    branches: [ main, master ]

jobs:
  build-android-apk:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source Code
        uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Setup Java JDK 21 for Android SDK
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '21'

      - name: Install Dependencies & Build Bundle
        run: |
          npm install
          npm run build
          npx --yes @capacitor/cli@latest init "TwinPing NOC" "com.twinping.noc.tablet" --web-dir dist || true
          npm install @capacitor/core @capacitor/android
          npx cap add android || true
          npx cap sync android

      - name: Compile Standalone Android APK (Gradle AssembleDebug)
        working-directory: ./android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Upload Ready-to-Install APK File
        uses: actions/upload-artifact@v4
        with:
          name: TwinPing-NOC-Android-Tablet-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk

#!/usr/bin/env bash
set -e
echo "======================================================"
echo "  TWINPING NOC - TRÌNH ĐÓNG GÓI FILE .APK ĐỘC LẬP"

chmod +x gradlew
./gradlew assembleDebug

echo "   android/app/build/outputs/apk/debug/app-debug.apk"
